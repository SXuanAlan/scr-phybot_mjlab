\section*{1\quad DeepMimic 模仿项（exp 核，参考量 $\hat\cdot$ 取自动作参考 clip 相位帧 $k$）}
\begin{align*}
r_{\mathrm{jpos}} &= \exp\!\Big(-\tfrac{1}{\sigma^2}\,\mathrm{mean}_j\big(\hat q_j - q_j\big)^2\Big),
  & \sigma &= 0.2\ \mathrm{rad} \w{2.0}\\[2pt]
r_{\mathrm{jvel}} &= \exp\!\Big(-\tfrac{1}{\sigma^2}\,\mathrm{mean}_j\big(\hat{\dot q}_j - \dot q_j\big)^2\Big),
  & \sigma &= 1.5\ \mathrm{rad/s} \w{0.5}\\[2pt]
r_{\mathrm{ee}} &= \exp\!\Big(-\tfrac{1}{\sigma^2}\,\mathrm{mean}_h\,
  \big\lVert \hat{\bm p}^{\,r}_h - \bm p^{\,r}_h \big\rVert^2\Big),
  & \sigma &= 0.1\ \mathrm{m} \w{1.5}\\[2pt]
r_{\mathrm{ori}} &= \exp\!\Big(-\tfrac{1}{\sigma^2}\,
  \theta_{\mathrm{geo}}\big(\hat q^{\mathrm{root}}, q^{\mathrm{root}}\big)^2\Big),
  & \sigma &= 0.3\ \mathrm{rad} \w{0.5}
\end{align*}
$\bm p^{\,r}_h$：EE 在\textbf{根系}的位置（双侧根相对 $\Rightarrow$ 只塑形肢体几何，不锚定全局轨迹）；
$\theta_{\mathrm{geo}}$：四元数测地角误差。

\section*{2\quad 相位啮合调度 与 爬升收入}
\begin{align*}
r_{\mathrm{mismatch}} &= \textstyle\sum_{h=1}^{4} \mathbf{1}\!\left[\hat e_h(k) \ne e_h\right]
  \w{-0.25}\\[6pt]
r_{\mathrm{height}} &= \mathrm{clip}\!\left(
  \frac{\big(\max(z^{\mathrm{best}},\,z_t) - z^{\mathrm{best}}\big)}
       {v_z^{\mathrm{cmd}}\,\Delta t},\ 0,\ 12\right),
  \qquad z^{\mathrm{best}} \leftarrow \max(z^{\mathrm{best}}, z_t) \w{2.0}\\[6pt]
r_{\mathrm{signed}} &= \mathrm{clip}\!\left(
  \frac{z_t - z_{t-1}}{v_z^{\mathrm{cmd}}\,\Delta t},\ -12,\ +12\right) \w{0.5}
\end{align*}
$\hat e_h(k)$：clip 第 $k$ 帧的啮合调度；$e_h$：实际 latch。
高水位项 settle 0.5\,s 内只跟随不计费；两项仅在 $|v_z^{\mathrm{cmd}}|>0.01$ 时激活。\\[4pt]
{\color{red}注：上限 12 倍参考速度且无逐 episode 预算帽 —— motionref1 ``撕爪狂奔''
（height:模仿收入 15:1）的直接原因；motionref2 起加 $1.25\times$ 参考里程预算帽。}

\section*{3\quad 姿态 / 几何护栏}
\begin{align*}
r_{\mathrm{lat}} &= \min\!\left(\big(v_y/0.1\big)^2,\ 4\right) \w{-0.2}\\[4pt]

  \frac{\big(|d - 0.196| - 0.08\big)_+}{0.1}\right)^{\!2},\ 4\right),

\section*{4\quad Microspine 物理契约}
\begin{align*}
r_{\mathrm{support}} &= \max\!\big(0,\ 2 - \textstyle\sum_h e_h\big) \w{-4.0}\\[6pt]
r_{\mathrm{release}} &= \frac{1}{\Delta t}\left[
  \sum_h \mathrm{rel}_h\, c(\mathrm{reason}_h)
  \;+\; \mathbf{1}\!\Big[\exists\,\mathrm{rel} \wedge \textstyle\sum_h e_h < 2\Big]\cdot 1.0
  \right] \w{-1.0}\\[2pt]
&\qquad c = \{\text{reverse}:0,\ \text{tension}:0.25,\ \text{lateral}:0.25,\ \text{ultimate}:1.0\}\\[6pt]
r_{\mathrm{slide}} &= \sum_h e_h\,\big\lVert \bm v_{h,\parallel} \big\rVert,
  \qquad \bm v_{h,\parallel} = \bm v_h - (\bm v_h\!\cdot\!\bm n)\,\bm n \w{-0.1}
\end{align*}
释放罚 $\div\Delta t$ 抵消 manager 的 $\times\Delta t$ $\Rightarrow$ 每事件一次性成本；
reverse（反耙脱钩）免费 = 预期的主动释放路径；settle 0.5\,s 免罚。
